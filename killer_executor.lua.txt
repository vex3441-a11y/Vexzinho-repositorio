-- Função de notificação
local function notify(title, text, duration)
    game.StarterGui:SetCore("SendNotification", {
        Title = title;
        Text = text;
        Duration = duration or 3;
    })
end

-- Atualizar tempo de sessão
spawn(function()
    while wait(1) do
        local elapsed = tick() - SessionStart
        local hours = math.floor(elapsed / 3600)
        local minutes = math.floor((elapsed % 3600) / 60)
        local seconds = math.floor(elapsed % 60)
        SessionTimeLabel.Text = string.format("⏱️ Sessão: %02d:%02d:%02d", hours, minutes, seconds)
    end
end)

-- Função para animar botões
local function animateButton(button, scale)
    local tween = TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = button.Size + UDim2.new(0, scale, 0, scale)
    })
    tween:Play()
    wait(0.2)
    local tween2 = TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = button.Size - UDim2.new(0, scale, 0, scale)
    })
    tween2:Play()
end

-- Função de análise de backdoors
local function analyzeBackdoors(script)
    local backdoorPatterns = {
        "require%(%d+%)",
        "loadstring",
        "getfenv",
        "setfenv",
        "HttpGet",
        "HttpPost",
        "game%.HttpGet",
        "game:HttpGet",
        "game:GetObjects",
        "InsertService",
        "MarketplaceService:GetProductInfo",
        "_G%.",
        "shared%."
    }
    
    local foundBackdoors = {}
    
    for _, pattern in ipairs(backdoorPatterns) do
        if string.find(script:lower(), pattern:lower()) then
            table.insert(foundBackdoors, pattern)
        end
    end
    
    return foundBackdoors
end

-- Função para atualizar status
local function updateStatus(text, color)
    StatusLabel.Text = text
    StatusIndicator.BackgroundColor3 = color
    
    local tween = TweenService:Create(StatusIndicator, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 20, 0, 20)
    })
    tween:Play()
    wait(0.5)
    local tween2 = TweenService:Create(StatusIndicator, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 16, 0, 16)
    })
    tween2:Play()
end

-- Botão Analisar
AnalyzeBtn.MouseButton1Click:Connect(function()
    animateButton(AnalyzeBtn, 5)
    
    local script = ScriptBox.Text
    if script == "" then
        updateStatus("❌ Nenhum script para analisar!", Color3.fromRGB(200, 50, 50))
        notify("⚠️ AVISO", "Cole um script primeiro!", 3)
        return
    end
    
    updateStatus("🔍 Analisando backdoors...", Color3.fromRGB(255, 200, 50))
    wait(1.5)
    
    local backdoors = analyzeBackdoors(script)
    
    if #backdoors > 0 then
        updateStatus("⚠️ " .. #backdoors .. " backdoor(s) detectado(s)!", Color3.fromRGB(255, 150, 50))
        notify("🔍 ANÁLISE", "Encontrados " .. #backdoors .. " backdoors potenciais!", 5)
        
        local backdoorList = "Backdoors encontrados:\n"
        for i, bd in ipairs(backdoors) do
            backdoorList = backdoorList .. i .. ". " .. bd .. "\n"
        end
        print(backdoorList)
    else
        updateStatus("✅ Nenhum backdoor detectado!", Color3.fromRGB(50, 200, 50))
        notify("✅ SEGURO", "Script não contém backdoors conhecidos!", 4)
    end
end)

-- Botão Executar
ExecuteBtn.MouseButton1Click:Connect(function()
    animateButton(ExecuteBtn, 5)
    
    local script = ScriptBox.Text
    if script == "" then
        updateStatus("❌ Nenhum script para executar!", Color3.fromRGB(200, 50, 50))
        notify("⚠️ AVISO", "Cole um script primeiro!", 3)
        return
    end
    
    -- Primeiro analisa
    local backdoors = analyzeBackdoors(script)
    
    if #backdoors > 0 then
        updateStatus("⚠️ Executando com backdoors...", Color3.fromRGB(255, 150, 50))
        notify("⚠️ ATENÇÃO", "Script contém backdoors! Executando mesmo assim...", 4)
    else
        updateStatus("▶ Executando script...", Color3.fromRGB(50, 200, 50))
    end
    
    wait(0.5)
    
    local success, error = pcall(function()
        loadstring(script)()
    end)
    
    if success then
        updateStatus("✅ Script executado com sucesso!", Color3.fromRGB(50, 200, 50))
        notify("✅ SUCESSO", "Script executado!", 3)
    else
        updateStatus("❌ Erro na execução!", Color3.fromRGB(200, 50, 50))
        notify("❌ ERRO", tostring(error), 5)
    end
end)
